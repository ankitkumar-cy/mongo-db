Select specific fields
db.studentsAgg.aggregate([
    {
        $project: {
            _id: 0,
            name: 1,
            department: 1,
            cgpa: 1
        }
    }
])

Show only first 5 students
db.studentsAgg.aggregate([
    {
        $limit: 5
    }
])

Skip first 3 students
db.studentsAgg.aggregate([
    {
        $skip: 3
    }
])

Add a calculated field
db.studentsAgg.aggregate([
    {
        $addFields: {
            cgpaPercentage: { $multiply: ["$cgpa", 9.5] }
        }
    }
])

Add a new field
db.studentsAgg.aggregate([
    {
        $set: {
            studentType: "Regular"
        }
    }
])

Replace the document structure
db.studentsAgg.aggregate([
    {
        $replaceWith: {
            name: "$name",
            cgpa: "$cgpa"
        }
    }
])

Select random students
db.studentsAgg.aggregate([
    {
        $sample: {
            size: 3
        }
    }
])

Save aggregation result into another collection
db.studentsAgg.aggregate([
    {
        $match: {
            cgpa: { $gte: 8.5 }
        }
    },
    {
        $out: "highCGPAStudents"
    }
])

Perform multiple aggregations at once
db.studentsAgg.aggregate([
    {
        $facet: {
            totalStudents: [
                { $count: "count" }
            ],
            averageCGPA: [
                {
                    $group: {
                        _id: null,
                        average: { $avg: "$cgpa" }
                    }
                }
            ]
        }
    }
])
